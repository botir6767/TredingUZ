const DATABASE_URL = "BU_YERGA_NEON_URL";

// ===== FOYDALANUVCHI QO'SHISH =====
export async function foydalanuvchiQosh(ism, email, parol) {
    try {
        const javob = await fetch(`${DATABASE_URL}/foydalanuvchilar`, {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({
                ism: ism,
                email: email,
                parol: parol,
                yaratilgan: new Date().toISOString()
            })
        });
        return await javob.json();
    } catch (xato) {
        console.error("Baza xatosi:", xato);
        return { success: false, message: xato.message };
    }
}

// ===== FOYDALANUVCHINI TOPISH =====
export async function foydalanuvchiTop(email) {
    try {
        const javob = await fetch(`${DATABASE_URL}/foydalanuvchilar?email=${email}`);
        const malumot = await javob.json();
        return malumot[0] || null;
    } catch (xato) {
        console.error("Baza xatosi:", xato);
        return null;
    }
}
